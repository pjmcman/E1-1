$ mkdir -p perm_test && touch perm_test/secret.txt

$ ls -ld perm_test perm_test/secret.txt

drwxr-xr-x 1 ADMIN 197121 0 Nov  8 22:29 perm_test
-rw-r--r-- 1 ADMIN 197121 0 Nov  8 22:29 perm_test/secret.txt
